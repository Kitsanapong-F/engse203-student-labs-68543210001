export async function updateStatus(id, status) {

  const newStatus = {

    status: status.status,

  };
  const found = requests.find((r) => r.id === id);

  await persist();

  return found ? structuredClone(found.push(newStatus)) : null;

} ถูกไหมอย่าพึ่งเฉลย
